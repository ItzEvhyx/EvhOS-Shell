cmake_minimum_required(VERSION 3.21)
project(EvhNotes LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 6.7 REQUIRED COMPONENTS Quick Qml Gui QuickPrivate)
add_library(evhnotesplugin MODULE NotesPlugin.cpp RichDocument.h)
target_link_libraries(evhnotesplugin PRIVATE Qt6::QuickPrivate Qt6::Qml Qt6::Gui)
set_target_properties(evhnotesplugin PROPERTIES LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/RichNotes")
configure_file(qmldir "${CMAKE_BINARY_DIR}/RichNotes/qmldir" COPYONLY)
add_executable(notes-document-test ../tests/document_test.cpp RichDocument.h)
target_link_libraries(notes-document-test PRIVATE Qt6::QuickPrivate Qt6::Qml Qt6::Gui)
enable_testing()
add_test(NAME notes-document COMMAND notes-document-test)
set_tests_properties(notes-document PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
